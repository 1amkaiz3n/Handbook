## 📌 2. CONTEXT CLASSIFIER 

```javascript
console.log('=== CONTEXT CLASSIFICATION ===')

function getContext(el) {
  if (!el) return 'unknown'

  if (el.tagName === 'SCRIPT') return 'SCRIPT_CONTEXT (CRITICAL)'
  if (el.tagName === 'STYLE') return 'STYLE_CONTEXT'
  if (el.tagName === 'INPUT' || el.tagName === 'TEXTAREA') return 'FORM_CONTEXT'

  // attribute-based risk
  const riskyAttrs = ['href', 'src', 'action', 'formaction', 'data', 'onclick', 'onerror']
  for (let attr of el.attributes || []) {
    if (riskyAttrs.includes(attr.name.toLowerCase())) {
      return 'ATTRIBUTE_CONTEXT (HIGH RISK)'
    }
  }

  return 'TEXT_CONTEXT (LOW/MED)'
}

// scan DOM
document.querySelectorAll('*').forEach(el => {
  const ctx = getContext(el)
  if (ctx.includes('CRITICAL') || ctx.includes('HIGH')) {
    console.log(`[CTX] ${ctx}`, el)
  }
})
```

## 📌 3. TRUE SINK DETECTOR (bukan sekadar keyword search)

```javascript
console.log('=== SINK DETECTION ENGINE ===')

const sinks = [
  { type: 'DOM_XSS', patterns: ['innerHTML', 'outerHTML', 'insertAdjacentHTML'] },
  { type: 'JS_EXEC', patterns: ['eval(', 'Function(', 'setTimeout(', 'setInterval('] },
  { type: 'URL_REDIRECT', patterns: ['location.href', 'location=', 'window.location'] }
]

document.querySelectorAll('script:not([src])').forEach(script => {
  const code = script.textContent

  sinks.forEach(sink => {
    sink.patterns.forEach(p => {
      if (code.includes(p)) {
        console.log(`[SINK FOUND] ${sink.type} → ${p}`)
        console.log(script)
      }
    })
  })
})
```

## 📌 6. HIGH VALUE DOM XSS SIGNAL SCANNER

```javascript
console.log('=== HIGH VALUE SIGNALS ===')

const signals = {
  url_usage: ['location.search', 'location.hash', 'URLSearchParams'],
  dangerous_dom: ['innerHTML', 'outerHTML'],
  execution: ['eval(', 'Function('],
  redirect: ['location.href', 'window.location']

  const code = script.textContent

  Object.entries(signals).forEach(([type, arr]) => {
    arr.forEach(sig => {

        console.log(`[SIGNAL] ${type} → ${sig}`)

      }
    })
  })
```
